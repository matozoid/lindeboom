[grammar]
grammar OpenDeviceStatement;
statement : 'OPEN' ( 'DEVICE' (  OPT1  |  OPT2  |  OPT3  )? )+ ;
program : statement+ '.' ;
OPT1 : 'OPT-1';
OPT2 : 'OPT-2';
OPT3 : 'OPT-3';
WS : (' '|'\n')+ -> channel(HIDDEN);

[start]
statement

[input]
"OPEN DEVICE DEVICE"

[tree]
(statement OPEN DEVICE DEVICE)
