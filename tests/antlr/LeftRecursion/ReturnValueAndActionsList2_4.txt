[grammar]
grammar T;
s  : expr EOF;
expr:
    a=expr '*' a=expr
    | b=expr ',' b=expr
    | b=expr '>>' c=expr
    | ID
   ;
ID  : ('a'..'z'|'A'..'Z'|'_')
      ('a'..'z'|'A'..'Z'|'0'..'9'|'_')*
;
WS : [ \t\n]+ -> skip ;

[start]
s

[input]
"a*b,c,x*y>>r"

[tree]
(s (expr (expr (expr (expr (expr a) * (expr b)) , (expr c)) , (expr (expr x) * (expr y))) >> (expr r)))
