[grammar]
grammar T;
s  : declarator EOF ;
declarator
        : declarator '[' e ']'
        | declarator '[' ']'
        | declarator '(' ')'
        | '*' declarator
        | '(' declarator ')'
        | ID
        ;
e : INT ;
ID : 'a'..'z'+ ;
INT : '0'..'9'+ ;
WS : (' '|'\n') -> skip ;

[start]
s

[input]
"b[]"

[tree]
(s (declarator (declarator b) [ ]))
